<script>
let pinUser=null;
function pinGate(user,msg){
  pinUser=user||pinUser;
  $('app').classList.add('hidden');
  $('auth').innerHTML='<div class="min-h-screen grid place-items-center p-5" style="background:#021812"><div class="card p-8 w-full max-w-md space-y-4"><h1 class="text-2xl font-extrabold text-center">Lecturer Access</h1><p class="text-xs text-slate-500 text-center">Enter the access PIN issued to you.</p><input id="l-pin" class="inp text-center font-mono font-bold uppercase" placeholder="ACCESS PIN" maxlength="12"><button class="btn w-full py-3" onclick="redeemPin()">Verify and Continue</button><button class="text-xs text-slate-500 w-full" onclick="logout()">Log out</button>'+(msg?'<p class="text-xs text-rose-500 text-center font-semibold">'+esc(msg)+'</p>':'')+'</div></div>';
}
async function redeemPin(){
  const pin=$('l-pin').value.trim();
  if(pin.length<4)return pinGate(null,'Enter your access PIN.');
  const{error}=await sb.rpc('redeem_lecturer_pin',{p_pin:pin});
  if(error)return pinGate(null,error.message);
  boot(pinUser);
}
</script>
